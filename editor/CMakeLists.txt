cmake_minimum_required(VERSION 3.25)
project(LykowareEditor VERSION 0.1.0 LANGUAGES CXX)
option(LYKO_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(lyko_core src/command_bus.cpp src/command_service.cpp src/scene_io.cpp src/mcp_server.cpp)
target_include_directories(lyko_core PUBLIC src)
target_compile_features(lyko_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(lyko_core PRIVATE /W4 /WX)
else()
  target_compile_options(lyko_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(LYKO_SANITIZERS)
    target_compile_options(lyko_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(lyko_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(lyko src/main.cpp)
target_link_libraries(lyko PRIVATE lyko_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(lyko_tests tests/command_bus_tests.cpp)
  target_link_libraries(lyko_tests PRIVATE lyko_core)
  add_test(NAME command_bus COMMAND lyko_tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME mcp_transport COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/mcp_tests.py $<TARGET_FILE:lyko>)
  add_test(NAME scene_persistence COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/persistence_tests.py $<TARGET_FILE:lyko>)
  add_test(NAME headless_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py $<TARGET_FILE:lyko>)
endif()
