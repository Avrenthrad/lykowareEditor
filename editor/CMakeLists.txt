cmake_minimum_required(VERSION 3.25)
project(LykowareEditor VERSION 0.1.0 LANGUAGES CXX)
option(LYKO_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(LYKO_THREAD_SANITIZER "Enable ThreadSanitizer (separate from ASan/UBSan)" OFF)
if(LYKO_SANITIZERS AND LYKO_THREAD_SANITIZER)
  message(FATAL_ERROR "ASan/UBSan and ThreadSanitizer must use separate build directories")
endif()
if(MSVC AND (LYKO_SANITIZERS OR LYKO_THREAD_SANITIZER))
  message(FATAL_ERROR "Sanitizer presets require GCC or Clang; use dev with MSVC")
endif()
option(LYKO_CLANG_TIDY "Run clang-tidy during compilation" OFF)
if(LYKO_CLANG_TIDY)
  find_program(LYKO_CLANG_TIDY_EXECUTABLE NAMES clang-tidy REQUIRED)
  set(CMAKE_CXX_CLANG_TIDY "${LYKO_CLANG_TIDY_EXECUTABLE};--warnings-as-errors=*")
endif()
find_package(Threads REQUIRED)
add_library(lyko_core src/asset_source.cpp src/assets.cpp src/prefab.cpp src/command_bus.cpp src/command_service.cpp src/scene_io.cpp src/mcp_server.cpp src/text_file.cpp src/ui/workspace_io.cpp src/ui/command_queue.cpp src/transport/mcp_stream.cpp)
target_link_libraries(lyko_core PUBLIC Threads::Threads)
target_include_directories(lyko_core PUBLIC src)
target_compile_features(lyko_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(lyko_core PRIVATE /W4 /WX)
else()
  target_compile_options(lyko_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(LYKO_SANITIZERS)
    target_compile_options(lyko_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(lyko_core PUBLIC -fsanitize=address,undefined)
  elseif(LYKO_THREAD_SANITIZER)
    target_compile_options(lyko_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(lyko_core PUBLIC -fsanitize=thread)
  endif()
endif()
add_executable(lyko src/main.cpp)
target_link_libraries(lyko PRIVATE lyko_core)
include(CTest)
